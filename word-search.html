/* Place words randomly, then fill gaps with random letters */
function buildGrid(words) {
let ok = false, tries = 0;
while (!ok && tries++ < 50) {
grid = Array.from({length: N}, () => Array(N).fill(""));
placed = [];
ok = true;
// longest words first = easier to fit
for (const w of words.slice().sort((a,b) => b.length - a.length)) {
if (!placeWord(w)) { ok = false; break; }
}
}
const A = "ABCDEFGHIJKLMNOPQRSTUVWXYZ";
for (let r = 0; r < N; r++)
for (let c = 0; c < N; c++)
if (!grid[r][c]) grid[r][c] = A[Math.floor(Math.random() * 26)];
}

function placeWord(word) {
for (let a = 0; a < 300; a++) {
const [dr, dc] = DIRS[Math.floor(Math.random() * DIRS.length)];
const r0 = Math.floor(Math.random() * N), c0 = Math.floor(Math.random() * N);
const r1 = r0 + dr * (word.length - 1), c1 = c0 + dc * (word.length - 1);
if (r1 < 0 || r1 >= N || c1 < 0 || c1 >= N) continue;

// check fit (empty or same letter)
let fits = true;
for (let i = 0; i < word.length; i++) {
const ch = grid[r0 + dr*i][c0 + dc*i];
if (ch && ch !== word[i]) { fits = false; break; }
}
if (!fits) continue;

const cells = [];
for (let i = 0; i < word.length; i++) {
const r = r0 + dr*i, c = c0 + dc*i;
grid[r][c] = word[i]; cells.push([r, c]);
}
placed.push({ word, cells, found: false, color: COLORS[placed.length % COLORS.length] });
return true;
}
return false;
}

RENDER

gridEl.innerHTML = "";
gridEl.style.setProperty("--n", N);
cellEls = [];
for (let r = 0; r < N; r++) {
cellEls.push([]);
for (let c = 0; c < N; c++) {
const d = document.createElement("div");
d.className = "cell"; d.textContent = grid[r][c];
d.dataset.r = r; d.dataset.c = c;
gridEl.appendChild(d); cellEls[r].push(d);
}
}
}

});
}

/* =====================================================
INTERACTION (drag + tap)
===================================================== */
function cellFromEvent(e) {
const el = document.elementFromPoint(e.clientX, e.clientY);
if (!el || !el.classList || !el.classList.contains("cell")) return null;
return [+el.dataset.r, +el.dataset.c];
}

function onDown(e) {
if (paused || gameOver) return;
const c = cellFromEvent(e); if (!c) return;
e.preventDefault();
try { gridEl.setPointerCapture(e.pointerId); } catch (_) {}
dragging = true; startCell = c;
// if a tap-start is pending, the line goes from tapStart to this cell
showSelection(tapStart ? line(tapStart, c) : [c]);
}

function onMove(e) {
if (!dragging) return;
const c = cellFromEvent(e); if (!c) return;
const from = tapStart && sameCell(startCell, tapStart) ? startCell : startCell;
showSelection(line(from, c));
}

function onUp(e) {
if (!dragging) return;
dragging = false;
const c = cellFromEvent(e) || startCell;

if (sameCell(startCell, c)) {
// ---- TAP MODE: first tap marks start, second tap marks end ----
if (!tapStart) { tapStart = startCell; showSelection([tapStart]); return; }
if (sameCell(tapStart, c)) { tapStart = null; clearSel(); return; } // tap same cell = cancel
const cells = line(tapStart, c);
tapStart = null; checkSelection(cells); return;
}
// ---- DRAG MODE ----
tapStart = null;
checkSelection(line(startCell, c));
}

function cancelSel() { dragging = false; tapStart = null; clearSel(); }

const sameCell = (a, b) => a && b && a[0] === b[0] && a[1] === b[1];

/* Build a straight line (H / V / diagonal) from a to b.
If b is not exactly on a line, snap to the nearest direction. */
function line(a, b) {
let dr = b[0] - a[0], dc = b[1] - a[1];
const adr = Math.abs(dr), adc = Math.abs(dc);
if (adr && adc && adr !== adc) {
if (adr > adc * 2) dc = 0; // mostly vertical
else if (adc > adr * 2) dr = 0; // mostly horizontal
else { const m = Math.round((adr + adc) / 2); // diagonal
dr = Math.sign(dr) * m; dc = Math.sign(dc) * m; }
}
const len = Math.max(Math.abs(dr), Math.abs(dc));
const sr = Math.sign(dr), sc = Math.sign(dc);
const cells = [];
for (let i = 0; i <= len; i++) {
const r = a[0] + sr*i, c = a[1] + sc*i;
if (r < 0 || r >= N || c < 0 || c >= N) break; // stay inside grid
cells.push([r, c]);
}
return cells;
}

function showSelection(cells) {
clearSel();
selCells = cells;
cells.forEach(([r, c]) => cellEls[r][c].classList.add("sel"));
}
function clearSel() {
selCells.forEach(([r, c]) => cellEls[r][c].classList.remove("sel"));
selCells = [];
}

/* Compare selected letters with unfound words (forward or reversed) */
function checkSelection(cells) {
const str = cells.map(([r, c]) => grid[r][c]).join("");
const rev = str.split("").reverse().join("");
const hit = placed.find(p => !p.found && (p.word === str || p.word === rev));
clearSel();
if (hit) markFound(hit);
}

/* =====================================================
FOUND WORD: glow, list strike, score
===================================================== */
function markFound(p) {
p.found = true; found++;
p.cells.forEach(([r, c], i) => {
const el = cellEls[r][c];
el.style.setProperty("--c", p.color);
el.classList.add("found");
setTimeout(() => { // staggered pop animation
el.classList.remove("pop"); void el.offsetWidth; el.classList.add("pop");
}, i * 55);
});
const w = $("w-" + p.word);
w.style.setProperty("--c", p.color); w.classList.add("done");

// score: 10 per letter + speed bonus (faster than 30s since last find)
const gap = seconds - lastFindSec; lastFindSec = seconds;
const bonus = Math.max(0, 30 - gap);
const gain = p.word.length * 10 + bonus;
score += gain;
toast(`+${gain} ${bonus ? "(⚡ bonus " + bonus + ")" : ""}`);
updateStats();

if (found === placed.length) setTimeout(victory, 700);
}

/* =====================================================
HINT: briefly flash first letter of a random hidden word
===================================================== */
function giveHint() {
if (paused || gameOver) return;
const left = placed.filter(p => !p.found);
if (!left.length) return;
const p = left[Math.floor(Math.random() * left.length)];
const [r, c] = p.cells[0];
const el = cellEls[r][c];
el.classList.remove("hint"); void el.offsetWidth; el.classList.add("hint");
setTimeout(() => el.classList.remove("hint"), 1300);
score = Math.max(0, score - 5); // small hint penalty
updateStats(); toast("💡 Hint (-5)");
}

function startTimer() {
clearInterval(timerId);
